func new_game():
	# clear all existing tasks
	for child in blocks_container.get_children():
		child.queue_free()
	
	#wait one frame for cleanup
	await get_tree().process_frame
	
	# recreate tasks that are on the right
	for i in range (9):
		var new_task = TASK.instantiate()
		new_task.task_done.connect(_on_task_completed)
		
		blocks_container.add_child(new_task)
		if i < 3:
			new_task.red()
			
		if i > 2 and i < 6:
			new_task.yellow()
		if i > 5:
			new_task.blue()
	
	for x in range (9):
		tiles[x].queue_free()
	tiles.clear()
	
	await get_tree().process_frame
	
	tile_color.shuffle()
		
	# Filling in the grid with blocks
	for i in range (9):
		var new_tile = TILE.instantiate()
		add_child(new_tile)
		new_tile.position = table_coordinates[i]
		new_tile.grid_id = i
		if tile_color[i] == 0:
			new_tile.red()
			new_tile.color = 0
		elif tile_color[i] == 1:
			new_tile.yellow()
			new_tile.color = 1
		else:
			new_tile.blue()
			new_tile.color = 2
		tiles.append(new_tile)
	show_game()
	return
